  <script>
    lucide.createIcons();

    const faixasCores = {
      "Branca": "bg-gray-300 text-gray-800",
      "Cinza": "bg-gray-500 text-white",
      "Amarela": "bg-yellow-400 text-black",
      "Laranja": "bg-orange-500 text-white",
      "Verde": "bg-green-500 text-white",
      "Azul": "bg-blue-600 text-white",
      "Roxa": "bg-purple-600 text-white",
      "Marrom": "bg-amber-800 text-white",
      "Preta": "bg-black text-white"
    };

    let atletas = [];
    const nomesExemplo = ["Ademir Antonio Frozza","Ally Francisco Varela Trevisan","Álvaro Araújo da Silva","Tomás Vicente Bonavigo","Anna Luiza Barbosa","Maria Eduarda de Melo Vieira","Aron Ferrari","Brayon Guimaraes Cavalcante"];
    const academias = ["The Match Champ","Araújo Jiu-Jitsu Team","Gracie Barra Palmas","EDI Team","Infantil A Brancacinza"];

    for (let i = 0; i < 152; i++) {
      const nome = nomesExemplo[Math.floor(Math.random()*nomesExemplo.length)] + " " + nomesExemplo[Math.floor(Math.random()*nomesExemplo.length)].split(" ")[0];
      const idade = 4 + Math.floor(Math.random()*14);
      const faixa = Object.keys(faixasCores)[Math.floor(Math.random()*Object.keys(faixasCores).length)];
      const peso = (20 + Math.random()*70).toFixed(2);
      const academia = academias[Math.floor(Math.random()*academias.length)];
      const pago = Math.random() > 0.25;
      const absoluto = Math.random() > 0.8;

      atletas.push({nome, idade, faixa, peso, academia, pago, absoluto});
    }

    const porPagina = 10;
    let paginaAtual = 1;

    function renderizar() {
      const inicio = (paginaAtual-1)*porPagina;
      const fim = inicio + porPagina;
      const pagina = atletas.slice(inicio, fim);

      document.getElementById('lista-atletas').innerHTML = pagina.map((a, i) => `
        <div class="bg-gray-50/70 hover:bg-gray-100 rounded-2xl p-8 flex items-center gap-6 border border-gray-200 transition-shadow hover:shadow-md">
          <div class="w-16 h-16 rounded-full bg-gray-800 text-white flex items-center justify-center font-bold text-2xl flex-shrink-0">
            ${a.nome.split(" ").map(n=>n[0]).join("")}
          </div>
          <div class="flex-1">
            <div class="flex items-center gap-4 mb-2">
              <p class="text-2xl font-bold text-gray-800">${a.nome.toUpperCase()}</p>
              ${a.absoluto ? '<span class="bg-orange-500 text-white px-4 py-1 rounded-full text-sm font-bold">ABSOLUTO</span>' : ''}
            </div>
            <p class="text-lg text-gray-600">${a.idade} anos • <span class="px-4 py-1 rounded-full font-bold text-sm ${faixasCores[a.faixa]}">${a.faixa}</span> • ${a.peso}kg</p>
            <p class="text-gray-500 mt-1">${a.academia}</p>
          </div>
          <div class="text-right">
            <p class="text-sm text-gray-500">ID: 0${1234 + inicio + i}</p>
            <p class="${a.pago ? 'text-green-600' : 'text-red-600'} font-bold text-lg flex items-center gap-2">
              <i data-lucide="${a.pago ? 'check-circle' : 'alert-circle'}"></i>
              ${a.pago ? 'PAGAMENTO CONFIRMADO' : 'PAGAMENTO PENDENTE'}
            </p>
          </div>
        </div>
      `).join('');

      const totalPag = Math.ceil(atletas.length / porPagina);
      document.getElementById('pagina-info').textContent = `Página ${paginaAtual} de ${totalPag}`;
      document.getElementById('prev').disabled = paginaAtual === 1;
      document.getElementById('next').disabled = paginaAtual === totalPag;

      lucide.createIcons();
    }

    document.getElementById('prev').onclick = () => { paginaAtual--; renderizar(); };
    document.getElementById('next').onclick = () => { paginaAtual++; renderizar(); };

    document.getElementById('busca').addEventListener('input', e => {
      const termo = e.target.value.toLowerCase();
      const filtrados = atletas.filter(a => 
        a.nome.toLowerCase().includes(termo) || a.academia.toLowerCase().includes(termo)
      );
      atletas = termo ? filtrados : atletas.slice();
      paginaAtual = 1;
      renderizar();
    });

    renderizar();
  </script>
